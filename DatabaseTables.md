CREATE TABLE `OrganizationLocations` (`LocationID` INT AUTO_INCREMENT PRIMARY KEY COMMENT 'Unique identifier for each organization location.', `LocationManager` VARCHAR(100) COMMENT 'Name of the person responsible for managing this location.', `ContactNumber` VARCHAR(20) NOT NULL COMMENT 'Primary contact phone number for this location.', `ContactEmail` VARCHAR(100) NOT NULL COMMENT 'Primary email address for this location.', `FullAddress` TEXT NOT NULL COMMENT 'Complete postal address of the location.', `GoogleMapEmbedUrl` TEXT COMMENT 'Google Maps embed URL used to display this location on the website.', `LocationType` VARCHAR(30) NOT NULL COMMENT 'Category of the location such as Head Office, Branch Office, Warehouse, Studio, Service Center, or Registered Office.', `Status` ENUM('Active', 'Inactive') NOT NULL DEFAULT 'Active' COMMENT 'Indicates whether the location is currently active or inactive.', `CreatedAt` DATETIME NOT NULL COMMENT 'Date and time when this record was created.', `CreatedBy` VARCHAR(100) NOT NULL COMMENT 'Name or identifier of the user who created this record.', `UpdatedAt` DATETIME DEFAULT NULL COMMENT 'Date and time when this record was last updated.',`UpdatedBy` VARCHAR(100) DEFAULT NULL COMMENT 'Name or identifier of the user who last updated this record.') ENGINE=InnoDB;

CREATE TABLE `WebsiteGallery` (`GalleryImageID` INT NOT NULL AUTO_INCREMENT PRIMARY KEY COMMENT 'Unique identifier for each gallery image record.' , `ImageFileName` TEXT NOT NULL COMMENT 'Original or stored filename of the gallery image.' , `ImageUrl` TEXT NOT NULL COMMENT 'Relative or absolute URL/path where the gallery image is stored.' , `CreatedAt` DATETIME NOT NULL COMMENT 'Date and time when this gallery image was added.' , `CreatedBy` VARCHAR(20) NOT NULL COMMENT 'Name or identifier of the user who uploaded this gallery image.') ENGINE = InnoDB;

CREATE TABLE `WebsiteEventGallery` (`EventGalleryID` INT NOT NULL AUTO_INCREMENT PRIMARY KEY COMMENT 'Unique identifier for each event gallery image record.' , `ImageFileName` TEXT NOT NULL COMMENT 'Original or stored filename of the event image.' , `ImageUrl` TEXT NOT NULL COMMENT 'Relative or absolute URL/path where the event image is stored.' , `CreatedAt` DATETIME NOT NULL COMMENT 'Date and time when the event image was uploaded.' , `CreatedBy` VARCHAR(20) NOT NULL COMMENT 'Name or identifier of the user who uploaded the event image.') ENGINE = InnoDB;

CREATE TABLE `EventBookings` (`BookingID` INT NOT NULL AUTO_INCREMENT PRIMARY KEY COMMENT 'Primary key of the booking record' , `BookingNo` VARCHAR(20) NOT NULL UNIQUE COMMENT 'System generated unique booking number (e.g. BK202600001)' , `CustomerName` VARCHAR(50) NOT NULL COMMENT 'Name of the customer who requested the booking' , `MobileNumber` VARCHAR(20) NOT NULL COMMENT 'Customer contact mobile number' , `EventType` INT(3) NOT NULL COMMENT 'Type of event (Wedding, Birthday, DJ, Corporate, Temple, etc.)' , `EventDate` DATE NOT NULL COMMENT 'Event start date' , `EventEndDate` DATE NOT NULL COMMENT 'Event end date (same as EventDate for one-day events)' , `EventLocation` TEXT NOT NULL COMMENT 'Complete venue address where the event will be conducted' , `Notes` TEXT NOT NULL COMMENT 'Additional customer requirements or special instructions' , `BookingStatus` ENUM('Pending','Accepted','Rejected','Cancelled','Completed') NOT NULL DEFAULT 'Pending' COMMENT 'Current status of the booking request' , `CreatedAt` DATETIME NOT NULL COMMENT 'Date and time when the booking request was created' , `CreatedBy` VARCHAR(50) NOT NULL COMMENT 'User or system that created the booking' , `UpdatedAt` DATETIME NOT NULL COMMENT 'Date and time when the booking was last updated' , `UpdatedBy` VARCHAR(20) NOT NULL COMMENT 'User who last modified the booking') ENGINE = InnoDB;
ALTER TABLE `EventBookings` ADD INDEX(`EventDate`);
ALTER TABLE `EventBookings` ADD INDEX(`BookingStatus`);

CREATE TABLE `EventTypes` (`EventTypeID` INT NOT NULL AUTO_INCREMENT COMMENT 'Unique identifier for each event type.' , `EventTypeName` VARCHAR(100) NOT NULL COMMENT 'Name of the event type (e.g., Wedding, Reception, Temple Event).' , `EventShortKey` CHAR(3) NOT NULL COMMENT 'Unique 3-character code for the event type (e.g., WED, REC, TMP).' , `Status` ENUM('Active','Inactive') NOT NULL DEFAULT 'Active' COMMENT 'Indicates whether the event type is available for booking.' , `CreatedAt` DATETIME NOT NULL COMMENT 'Date and time when this record was created.' , `CreatedBy` VARCHAR(50) NOT NULL COMMENT 'User who created this record.' , `UpdatedAt` DATETIME NOT NULL COMMENT 'Date and time when this record was last updated.' , `UpdatedBy` VARCHAR(50) NOT NULL COMMENT 'User who last updated this record.' , PRIMARY KEY (`EventTypeID`), UNIQUE (`EventTypeName`), UNIQUE (`EventShortKey`)) ENGINE = InnoDB;

CREATE TABLE `EventCalendar` (`CalendarID` INT NOT NULL AUTO_INCREMENT , `EventDate` DATE NOT NULL , `BookingAvailability` ENUM('Available','PartiallyBooked','FullyBooked') NOT NULL , `Remarks` VARCHAR(150) NULL , `CreatedAt` DATETIME NOT NULL , `CreatedBy` VARCHAR(50) NOT NULL , `UpdatedAt` DATETIME NULL , `UpdatedBy` VARCHAR(50) NULL , PRIMARY KEY (`CalendarID`)) ENGINE = InnoDB;

CREATE TABLE `AdminLogins` (`AdminLoginID` INT NOT NULL AUTO_INCREMENT , `AdminUsername` VARCHAR(100) NOT NULL , `AdminPassword` VARCHAR(100) NOT NULL , `CreatedAt` DATETIME NOT NULL , `CreatedBy` VARCHAR(50) NOT NULL , PRIMARY KEY (`AdminLoginID`)) ENGINE = InnoDB;

CREATE TABLE `AdminLogTimings` (`LogTimingsID` INT NOT NULL AUTO_INCREMENT , `UserName` VARCHAR(50) NOT NULL , `LogStatus` ENUM('Login','Logout') NOT NULL , `LoginAt` DATETIME NULL DEFAULT NULL , `LogoutAt` DATETIME NULL DEFAULT NULL , `CreatedAt` DATETIME NOT NULL , PRIMARY KEY (`LogTimingsID`)) ENGINE = InnoDB;

CREATE TABLE `AdminSessions` (`SessionID` INT NOT NULL AUTO_INCREMENT , `AdminLoginID` INT NOT NULL , `Token` TEXT NOT NULL , `DeviceInfo` VARCHAR(200) NOT NULL , `IPAddress` VARCHAR(50) NOT NULL , `IsActive` TINYINT(1) NOT NULL , `CreatedAt` DATETIME NOT NULL , `ExpiresAt` DATETIME NULL DEFAULT NULL , PRIMARY KEY (`SessionID`)) ENGINE = InnoDB;
